// ── STORY VIEWER ──────────────────────────────────────────────────────────────
function StoryViewer({ story, onClose, onReply, allStories, setStoryIdx }) {
  const T = useTheme();
  const [slideIdx, setSlideIdx] = useState(0);
  const [replyText, setReplyText] = useState("");
  const [replySent, setReplySent] = useState(false);
  const slide = story.slides[slideIdx];
  const progress = ((slideIdx + 1) / story.slides.length) * 100;

  const goNext = () => {
    if (slideIdx < story.slides.length - 1) setSlideIdx(i => i + 1);
    else {
      const nextIdx = allStories.findIndex(s => s.id === story.id) + 1;
      if (nextIdx < allStories.length) { setStoryIdx(nextIdx); setSlideIdx(0); }
      else onClose();
    }
  };
  const goPrev = () => {
    if (slideIdx > 0) setSlideIdx(i => i - 1);
    else {
      const prevIdx = allStories.findIndex(s => s.id === story.id) - 1;
      if (prevIdx >= 0) { setStoryIdx(prevIdx); setSlideIdx(0); }
    }
  };
  const sendReply = () => {
    if (!replyText.trim()) return;
    onReply(story.user, replyText.trim());
    setReplySent(true);
    setReplyText("");
    setTimeout(() => setReplySent(false), 2000);
  };

  return (
    <div style={{ position: "fixed", inset: 0, zIndex: 999, background: "#000", display: "flex", flexDirection: "column", maxWidth: 430, margin: "0 auto" }}>
      {/* Progress bars */}
      <div style={{ display: "flex", gap: 4, padding: "12px 12px 8px" }}>
        {story.slides.map((_, i) => (
          <div key={i} style={{ flex: 1, height: 3, borderRadius: 2, background: "rgba(255,255,255,0.3)", overflow: "hidden" }}>
            <div style={{ height: "100%", background: "#fff", width: i < slideIdx ? "100%" : i === slideIdx ? "60%" : "0%", transition: i === slideIdx ? "width 5s linear" : "none" }} />
          </div>
        ))}
      </div>

      {/* Header */}
      <div style={{ display: "flex", alignItems: "center", gap: 10, padding: "0 14px 12px" }}>
        <div style={{ width: 36, height: 36, borderRadius: "50%", background: T.primary, display: "flex", alignItems: "center", justifyContent: "center", color: "#fff", fontWeight: 800, fontSize: 13, border: "2px solid #fff" }}>{story.avatar}</div>
        <span style={{ fontWeight: 700, fontSize: 14, color: "#fff" }}>{story.user}</span>
        <span style={{ fontSize: 12, color: "rgba(255,255,255,0.6)" }}>now</span>
        <button onClick={onClose} style={{ marginLeft: "auto", background: "none", border: "none", color: "#fff", fontSize: 22, cursor: "pointer" }}>✕</button>
      </div>

      {/* Story slide */}
      <div style={{ flex: 1, background: slide.bg, display: "flex", flexDirection: "column", alignItems: "center", justifyContent: "center", position: "relative" }}>
        <div style={{ fontSize: 80, marginBottom: 20 }}>{slide.emoji}</div>
        <p style={{ color: "#fff", fontSize: 18, fontWeight: 700, textAlign: "center", padding: "0 24px" }}>{slide.caption}</p>

        {/* Tap zones */}
        <div onClick={goPrev} style={{ position: "absolute", left: 0, top: 0, bottom: 0, width: "40%", cursor: "pointer" }} />
        <div onClick={goNext} style={{ position: "absolute", right: 0, top: 0, bottom: 0, width: "40%", cursor: "pointer" }} />
      </div>

      {/* Reply bar */}
      {replySent ? (
        <div style={{ padding: "14px 16px 28px", background: "rgba(0,0,0,0.8)", textAlign: "center" }}>
          <p style={{ color: T.primary, fontWeight: 700, fontSize: 14 }}>✓ Reply sent — check your DMs!</p>
        </div>
      ) : (
        <div style={{ padding: "10px 14px 28px", background: "rgba(0,0,0,0.7)", display: "flex", gap: 10, alignItems: "center" }}>
          <input
            value={replyText}
            onChange={e => setReplyText(e.target.value)}
            onKeyDown={e => e.key === "Enter" && sendReply()}
            placeholder={`Reply to ${story.user}...`}
            style={{ flex: 1, padding: "10px 14px", borderRadius: 22, border: "1.5px solid rgba(255,255,255,0.3)", background: "rgba(255,255,255,0.1)", color: "#fff", fontSize: 14, outline: "none" }}
          />
          <button onClick={sendReply} style={{ width: 38, height: 38, borderRadius: "50%", border: "none", background: replyText.trim() ? T.primary : "rgba(255,255,255,0.2)", color: "#fff", fontSize: 16, cursor: "pointer" }}>➤</button>
        </div>
      )}
    </div>
  );
}

function ChatScreen({ convo, onBack, storyReplies = {} }) {
  const T = useTheme();
  const baseHistory = CONVO_HISTORY[convo.name] || [];
  const extraReplies = storyReplies[convo.user] || storyReplies[convo.name] || [];
  const [messages, setMessages] = useState([...baseHistory, ...extraReplies]);
  const [input, setInput] = useState("");
  const bottomRef = { current: null };

  const send = () => {
    const text = input.trim();
    if (!text) return;
    setMessages(m => [...m, { from: "me", text, time: "now" }]);
    setInput("");
    // Auto-reply after 1s
    setTimeout(() => {
      const replies = ["😂 haha", "That's fire 🔥", "Bet 💪", "Say less", "Facts bro", "Lmaooo", "Let's gooo 🎉", "100%", "Ok ok I see you 👀", "Respect 🤝"];
      setMessages(m => [...m, { from: "them", text: replies[Math.floor(Math.random() * replies.length)], time: "now" }]);
    }, 1000);
  };

  return (
    <div style={{ position: "absolute", inset: 0, display: "flex", flexDirection: "column", background: T.bg }}>
      {/* Header */}
      <div style={{ flexShrink: 0, display: "flex", alignItems: "center", gap: 12, padding: "14px 16px", background: "#fff", borderBottom: "1px solid #E5E7EB", zIndex: 10 }}>
        <button onClick={onBack} style={{ background: "none", border: "1.5px solid #E5E7EB", borderRadius: 10, padding: "5px 10px", cursor: "pointer", fontSize: 16 }}>←</button>
        <div style={{ width: 40, height: 40, borderRadius: "50%", background: T.primary, display: "flex", alignItems: "center", justifyContent: "center", color: "#fff", fontWeight: 800, fontSize: 15 }}>{convo.name[0]}</div>
        <div>
          <p style={{ fontWeight: 800, fontSize: 15, color: "#1A1A2E" }}>{convo.name}</p>
          <p style={{ fontSize: 11, color: T.primary, fontWeight: 600 }}>● Online</p>
        </div>
        <div style={{ marginLeft: "auto", display: "flex", gap: 12 }}>
          <span style={{ fontSize: 20, cursor: "pointer" }}>📞</span>
          <span style={{ fontSize: 20, cursor: "pointer" }}>📹</span>
        </div>
      </div>

      {/* Messages */}
      <div style={{ flex: 1, overflowY: "auto", minHeight: 0, padding: "16px", display: "flex", flexDirection: "column", gap: 8 }}>
        {messages.map((msg, i) => {
          const isMe = msg.from === "me";
          const showTime = i === 0 || messages[i - 1].time !== msg.time;
          return (
            <div key={i}>
              {showTime && i % 5 === 0 && (
                <p style={{ textAlign: "center", fontSize: 11, color: "#aaa", margin: "8px 0", fontWeight: 500 }}>{msg.time}</p>
              )}
              <div style={{ display: "flex", justifyContent: isMe ? "flex-end" : "flex-start" }}>
                {!isMe && (
                  <div style={{ width: 28, height: 28, borderRadius: "50%", background: T.primary, display: "flex", alignItems: "center", justifyContent: "center", color: "#fff", fontWeight: 800, fontSize: 11, marginRight: 8, flexShrink: 0, alignSelf: "flex-end" }}>{convo.name[0]}</div>
                )}
                <div style={{
                  maxWidth: "72%",
                  padding: "10px 14px",
                  borderRadius: isMe ? "18px 18px 4px 18px" : "18px 18px 18px 4px",
                  background: isMe ? T.primary : "#fff",
                  color: isMe ? "#fff" : "#1A1A2E",
                  fontSize: 14,
                  lineHeight: 1.45,
                  fontWeight: 500,
                  boxShadow: isMe ? "none" : "0 1px 4px rgba(0,0,0,0.06)",
                  border: isMe ? "none" : "1px solid #E5E7EB",
                }}>
                  {msg.text}
                </div>
              </div>
            </div>
          );
        })}
        <div ref={el => { bottomRef.current = el; }} />
      </div>

      {/* Input bar */}
      <div style={{ flexShrink: 0, padding: "10px 14px 20px", background: "#fff", borderTop: "1px solid #E5E7EB", display: "flex", gap: 10, alignItems: "center", zIndex: 10 }}>
        <button style={{ fontSize: 22, background: "none", border: "none", cursor: "pointer", flexShrink: 0 }}>📎</button>
        <input
          value={input}
          onChange={e => setInput(e.target.value)}
          onKeyDown={e => e.key === "Enter" && send()}
          placeholder="Message..."
          style={{ flex: 1, padding: "11px 14px", borderRadius: 22, border: "1.5px solid #E5E7EB", fontSize: 14, outline: "none", background: "#F7F8FA" }}
        />
        <button
          onClick={send}
          style={{ width: 40, height: 40, borderRadius: "50%", border: "none", background: input.trim() ? T.primary : "#E5E7EB", color: "#fff", fontSize: 18, cursor: "pointer", display: "flex", alignItems: "center", justifyContent: "center", flexShrink: 0, transition: "background 0.15s" }}

export default function App() {
  const [screen, setScreen]       = useState("splash");
  const [subScreen, setSubScreen] = useState(null);
  const [palette, setPalette]     = useState("default");
  const [storyReplies, setStoryReplies] = useState({});
  const T = PALETTES[palette];

  const mainScreens = ["home", "social", "profile", "schedule", "nutrition", "pumpmatch", "shop"];
  const isMain = mainScreens.includes(screen);
  const navScreens = ["home", "social", "profile"];

  const renderMain = () => {
    if (subScreen === "dms") return <DMsSubScreen onBack={() => setSubScreen(null)} storyReplies={storyReplies} />;
    switch (screen) {
      case "home":      return <HomeScreen      setScreen={setScreen} />;
      case "social":    return <SocialScreen    setSubScreen={setSubScreen} storyReplies={storyReplies} setStoryReplies={setStoryReplies} />;
      case "schedule":  return <ScheduleScreen  onBack={() => setScreen("home")} />;
      case "nutrition": return <NutritionScreen subScreen={subScreen} setSubScreen={setSubScreen} onBack={() => setScreen("home")} />;
      case "pumpmatch": return <PumpMatchScreen onBack={() => setScreen("home")} />;
      case "shop":      return <ShopScreen      onBack={() => setScreen("home")} />;
      case "profile":   return <ProfileScreen   setScreen={setScreen} palette={palette} setPalette={setPalette} />;
      default:          return null;
    }
  };

  return (
    <ThemeCtx.Provider value={T}>
      <div style={{ maxWidth: 430, margin: "0 auto", minHeight: "100vh", display: "flex", flexDirection: "column", fontFamily: "-apple-system, BlinkMacSystemFont, 'Inter', 'Segoe UI', sans-serif", background: T.bg }}>
        {screen === "splash"     && <SplashScreen next={() => setScreen("signin")} />}
        {screen === "signin"     && <SignInScreen  setScreen={setScreen} />}
        {screen === "signup"     && <SignUpScreen  setScreen={setScreen} />}
        {screen === "onboarding" && <OnboardingScreen setScreen={setScreen} palette={palette} setPalette={setPalette} />}
        {isMain && (
          <>
            <div style={{ flex: 1, display: "flex", flexDirection: "column", overflow: "hidden", position: "relative" }}>
              {renderMain()}
            </div>
            {!subScreen && navScreens.includes(screen) && (
              <NavBar active={screen} setScreen={(s) => { setSubScreen(null); setScreen(s); }} />
            )}
          </>
        )}
      </div>
    </ThemeCtx.Provider>
  );
}
